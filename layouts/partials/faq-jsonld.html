{{/* FAQ sections are H2 headings; questions are H3 headings. Split sections first so the next section heading never becomes part of the preceding answer. Strip explicit Markdown heading IDs from schema question names. */}} {{- $items := slice -}} {{- range $section := after 1 (split .RawContent "\n## ") -}} {{- range $question := after 1 (split $section "\n### ") -}} {{- $lines := split $question "\n" -}} {{- $rawQ := replaceRE "\\s*\\{#[^}]+\\}\\s*$" "" (index $lines 0) -}} {{- $q := trim (htmlUnescape (plainify (markdownify $rawQ))) " \n\r\t" -}} {{- $answer := trim (delimit (after 1 $lines) "\n") " \n\r\t" -}} {{- $answerHTML := trim (string (markdownify $answer)) " \n\r\t" -}} {{- if and $q $answerHTML -}} {{- $items = $items | append (dict "@type" "Question" "name" $q "acceptedAnswer" (dict "@type" "Answer" "text" $answerHTML) ) -}} {{- end -}} {{- end -}} {{- end -}} {{- $obj := dict "@context" "https://schema.org" "@type" "FAQPage" "mainEntity" $items -}}
